#include "get_next_line.h"

char	*get_next_line(int fd)
{
	char	buffer[BUFFER_SIZE];
	int		b;

	b = read(fd, buffer, 1);
	while (b > 0 && buffer[0] != '\n')
	{
		write(1, buffer, b);
		b = read(fd, buffer, 1);
	}
	return (0);
}
